\begin{lemma}[Conditional law of the reward]
  \label{lem:reward_cond_law}
  \lean{Learning.IsAlgEnvSeq.lintegral_preReward_reward}
  \leanok
  \uses{def:repeated_game}
  In a run of a player against an adaptive opponent with reward kernel $R$, the reward $r_t$ has
  conditional law $R_t(h_t, x_t, y_t)$ given the first $t$ rounds $h_t$ and the actions $(x_t, y_t)$
  of round $t$: $\E[F(h_t, x_t, y_t, r_t)] = \E[\int F(h_t, x_t, y_t, r)\, R_t(h_t, x_t, y_t)(dr)]$
  for every measurable $F \ge 0$.
\end{lemma}

\begin{proof}
  \leanok
  Given $(h_t, x_t)$, the feedback $(y_t, r_t)$ has the law of the game environment, the
  composition-product of the policy of the opponent (on the swapped history) and of the reward
  kernel; integrate against it in both expressions.
\end{proof}

\begin{lemma}[Bounded noise is sub-Gaussian]
  \label{lem:bounded_noise_subgaussian}
  \lean{Learning.RepeatedGame.RewardKernel.hasSubgaussianMGF_sub,
    Learning.IsAlgEnvSeq.isCondSubgaussianNoise_preRewardFiltration}
  \leanok
  \uses{def:repeated_game, lem:reward_cond_law}
  If the reward has conditional mean $u(x_t, y_t)$ and values in $[-1, 1]$, the noise
  $r_t - u(x_t, y_t)$ is conditionally $1$-sub-Gaussian given the past and $(x_t, y_t)$: for every
  nonnegative $g$ and real $c$ measurable with respect to the $\sigma$-algebra $\mathcal G_t$
  generated by $(h_t, x_t, y_t)$, $\E[g e^{c (r_t - u(x_t, y_t))}] \le \E[g e^{c^2/2}]$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:reward_cond_law}
  Hoeffding's lemma (Mathlib) for a centered variable in an interval of length $2$, applied to the
  reward kernel at each history and action pair; then Lemma~\ref{lem:reward_cond_law}, since $g$
  and $c$ are functions of $(h_t, x_t, y_t)$ (Doob--Dynkin).
\end{proof}

\begin{lemma}[Ville's inequality]
  \label{lem:ville}
  \lean{MeasureTheory.mul_measure_exists_le_le_lintegral,
    MeasureTheory.measure_exists_le_le_lintegral_div}
  \leanok
  Let $Z$ be a nonnegative (possibly infinite) process adapted to a filtration $(\mathcal F_n)$,
  with $\int_A Z_{n+1} \le \int_A Z_n$ for every $n$ and $A \in \mathcal F_n$. Then
  $c\, \Prob(\exists n, Z_n \ge c) \le \E[Z_0]$ for every $c \ge 0$. In particular, for a
  nonnegative supermartingale $Z$ with $Z_0 = 1$, $\Prob(\exists t, Z_t \ge 1/\delta) \le \delta$.
\end{lemma}

\begin{proof}
  \leanok
  With $B_N = \{Z_k < c \text{ for all } k < N\} \in \mathcal F_{N-1}$, induction on $N$ gives
  $c\, \Prob(\exists k < N, Z_k \ge c) + \int_{B_N} Z_N \le \E[Z_0]$: split $\int_{B_N} Z_N$ on
  $\{Z_N \ge c\}$ and apply the supermartingale inequality on $B_{N+1} \in \mathcal F_N$. Let
  $N \to \infty$ (monotone convergence of the events).
\end{proof}

\begin{lemma}[Gaussian integral of a quadratic form]
  \label{lem:gaussian_quadratic}
  \lean{Matrix.PosDef.lintegral_exp_inner_sub_mahalanobisSq}
  \leanok
  For a positive definite $V \in \R^{d \times d}$ and $s \in \R^d$,
  $\int_{\R^d} \exp(\langle \theta, s\rangle - \frac12 \theta^\top V \theta)\, d\theta =
  (2\pi)^{d/2} (\det V)^{-1/2} \exp(\frac12 s^\top V^{-1} s)$.
\end{lemma}

\begin{proof}
  \leanok
  Change of variables $x = V^{1/2} \theta$ (Jacobian $(\det V)^{1/2}$), which turns the integrand
  into $\exp(\langle x, V^{-1/2} s\rangle - \frac12 \norm{x}^2)$, and the isotropic Gaussian integral
  with a linear term (Mathlib).
\end{proof}

\begin{lemma}[Gaussian mixture supermartingale]
  \label{lem:mixture_supermartingale}
  \lean{Learning.setLIntegral_selfNormMixture_succ_le, Learning.selfNormMixture_zero,
    Learning.selfNormMixture}
  \leanok
  \uses{lem:gaussian_quadratic}
  Let $S_t = \sum_{s < t} \eps_s a_s$ with predictable $a_s \in \R^d$ and conditionally
  $1$-sub-Gaussian $\eps_s$, and $V_t = \lambda I + \sum_{s < t} a_s a_s^\top$, $\lambda > 0$. Then
  $\bar Z_t = (\lambda^d/\det V_t)^{1/2} \exp(\frac12 \norm{S_t}^2_{V_t^{-1}})$ is a nonnegative
  supermartingale with initial value $1$. In dimension one, with $\lambda = 1$ and
  $a_s = I_s \in \{0, 1\}$, it is $(1 + N_t)^{-1/2} \exp(\frac{M_t^2}{2(1 + N_t)})$ with
  $M_t = \sum_{s < t} I_s \eps_s$ and $N_t = \sum_{s < t} I_s$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:gaussian_quadratic}
  For each $\theta$, $W_t(\theta) = \exp(\langle \theta, S_t\rangle - \frac12 \norm{\theta}^2_{V_t})$
  satisfies $W_{t+1}(\theta) = W_t(\theta) \exp(\langle \theta, a_t\rangle \eps_t - \frac12
  \langle \theta, a_t\rangle^2)$, so it is a supermartingale by the conditional sub-Gaussianity of
  $\eps_t$ (with the predictable coefficient $\langle \theta, a_t\rangle$). By
  Lemma~\ref{lem:gaussian_quadratic}, $\int W_t(\theta)\, d\theta = (2\pi/\lambda)^{d/2} \bar Z_t$
  (this is the mixture over $\theta \sim \mathcal N(0, \lambda^{-1} I)$ of
  $\exp(\langle \theta, S_t\rangle - \frac12 \norm{\theta}^2_{V_t - \lambda I})$, with the Gaussian
  density absorbed in $W$), and by Tonelli's theorem the mixture of the supermartingales
  $W(\theta)$ is a supermartingale.
\end{proof}

\begin{lemma}[Self-normalized bound, Abbasi-Yadkori, Pál, Szepesvári]
  \label{lem:self_normalized}
  \lean{Learning.measure_exists_lt_mahalanobisSq_inv_regGram_le,
    Learning.probReal_forall_mahalanobisSq_inv_regGram_le_ge,
    Learning.IsAlgEnvSeq.probReal_forall_mahalanobisSq_inv_regGram_le_ge}
  \leanok
  \uses{lem:mixture_supermartingale, lem:ville}
  Let $S_t = \sum_{s < t} \eps_s a_s$ with predictable $a_s \in \R^d$ and conditionally
  $1$-sub-Gaussian $\eps_s$, $V_t = \lambda I + \sum_{s < t} a_s a_s^\top$, $\lambda > 0$. With
  probability at least $1 - \delta$, for all $t$,
  $\norm{S_t}^2_{V_t^{-1}} \le 2 \log(1/\delta) + \log(\det V_t/\lambda^d)$. In particular, this
  holds in a run of any player against any adaptive adversary, with reward noise of conditional
  mean $u$ and rewards in $[-1, 1]$, for the features $a_s = f(x_s, y_s)$ of the action pairs and
  the noise $\eps_s = r_s - u(x_s, y_s)$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:mixture_supermartingale, lem:ville, lem:bounded_noise_subgaussian}
  Ville's inequality (Lemma~\ref{lem:ville}) for the supermartingale $\bar Z$ of
  Lemma~\ref{lem:mixture_supermartingale}, with $\bar Z_0 = 1$: $\bar Z_t \ge 1/\delta$ as soon as
  $\norm{S_t}^2_{V_t^{-1}} > 2 \log(1/\delta) + \log(\det V_t/\lambda^d)$. For the game, the
  features are predictable and the noise is conditionally $1$-sub-Gaussian
  (Lemma~\ref{lem:bounded_noise_subgaussian}) for the filtration generated by the first $t$ rounds
  and the actions $(x_t, y_t)$.
\end{proof}

\begin{lemma}[Error of the ridge estimate]
  \label{lem:ridge_error}
  \lean{Bandits.Linear.sqrt_mahalanobisSq_sub_ridgeEstimate_le}
  \leanok
  For $\lambda > 0$, features $a_s$, rewards $r_s$ and any $\theta$, with
  $V_t = \lambda I + \sum_{s < t} a_s a_s^\top$, $b_t = \sum_{s < t} r_s a_s$ and
  $S_t = \sum_{s < t} (r_s - \langle a_s, \theta\rangle) a_s$,
  $\norm{\theta - V_t^{-1} b_t}_{V_t} \le \norm{S_t}_{V_t^{-1}} + \sqrt\lambda \norm{\theta}$.
\end{lemma}

\begin{proof}
  \leanok
  $b_t = (V_t - \lambda I)\theta + S_t$, so $\theta - V_t^{-1} b_t = V_t^{-1}(\lambda \theta - S_t)$
  and $\norm{V_t^{-1} w}_{V_t} = \norm{w}_{V_t^{-1}}$; the triangle inequality, and
  $\norm{\theta}^2_{V_t^{-1}} \le \norm{\theta}^2/\lambda$ since $V_t \succeq \lambda I$.
\end{proof}

\begin{lemma}[Determinant bound]
  \label{lem:log_det_le}
  \lean{Learning.log_det_regGram_le, Ito2026Adversarial.log_det_regGram_div_le}
  \leanok
  If $\norm{a_s} \le L$, then $\log \det V_t \le d \log(\lambda + L^2 t/d)$; for $L = 1$,
  $\log(\det V_t/\lambda^d) \le d \log(1 + t/(d\lambda))$.
\end{lemma}

\begin{proof}
  \leanok
  AM--GM on the eigenvalues of the positive definite $V_t$: $\det V_t \le (\operatorname{tr} V_t /
  d)^d$, and $\operatorname{tr} V_t = \lambda d + \sum_s \norm{a_s}^2$.
\end{proof}

\begin{lemma}[Matrix determinant lemma for the Gram matrix]
  \label{lem:det_regGram_succ}
  \lean{Learning.det_regGram_succ}
  \leanok
  For $\lambda > 0$, $\det V_{t+1} = \det V_t (1 + \norm{a_t}^2_{V_t^{-1}})$.
\end{lemma}

\begin{proof}
  \leanok
  $\det(V + a a^\top) = \det V (1 + a^\top V^{-1} a)$ for invertible $V$ (Schur complement).
\end{proof}

\begin{lemma}[Elliptical potential]
  \label{lem:elliptical_potential}
  \lean{Learning.sum_min_one_mahalanobisSq_inv_regGram_le,
    Learning.sum_min_one_mahalanobisSq_inv_regGram_le_card_mul_log}
  \leanok
  \uses{lem:log_det_le, lem:det_regGram_succ}
  For $\lambda > 0$, $\sum_{t < T} \min(1, \norm{a_t}^2_{V_t^{-1}}) \le 2 \log(\det V_T/\det V_0)$;
  if moreover $\norm{a_t} \le L$ for $t < T$, this is at most $2 d \log(1 + L^2 T/(d\lambda))$.
  (With $\lambda \ge L^2$ the minimum is $\norm{a_t}^2_{V_t^{-1}}$.)
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:log_det_le, lem:det_regGram_succ}
  By Lemma~\ref{lem:det_regGram_succ}, $\log \det V_{t+1} - \log \det V_t = \log(1 +
  \norm{a_t}^2_{V_t^{-1}})$, and $\min(1, x) \le \frac{2x}{1 + x} \le 2 \log(1 + x)$ for $x \ge 0$;
  telescope, then Lemma~\ref{lem:log_det_le} (with $\det V_0 = \lambda^d$).
\end{proof}
